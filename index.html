const PLANILHA_ID = "1XmtLh00v_kcUEk2iARyG5pcwsRQOMv2xRCBaig7Xm6k"; const ABA_CADASTRO = "CADASTRO_LOTES"; const ABA_BENEFICIAMENTOS = "BENEFICIAMENTOS"; // ============================================================ // API // ============================================================ function doGet(e) { try { const parametros = e && e.parameter ? e.parameter : {}; const acao = parametros.acao || ""; const qr = parametros.qr || ""; const lote = parametros.lote || ""; // -------------------------------------------------------- // TESTE DA API // -------------------------------------------------------- if (acao === "") { return respostaJSON({ status: "online", mensagem: "API AgroTrace funcionando" }); } // -------------------------------------------------------- // BUSCAR LOTE // -------------------------------------------------------- if (acao === "buscar") { const resultado = buscarLote(qr, lote); /* * IMPORTANTE: * Não dependemos mais de callback enviado pela URL. * * A resposta sempre chama: * agroTraceCallback(...) */ return ContentService .createTextOutput( "agroTraceCallback(" + JSON.stringify(resultado) + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } // -------------------------------------------------------- // AÇÃO DESCONHECIDA // -------------------------------------------------------- return respostaJSON({ status: "erro", mensagem: "Ação não reconhecida." }); } catch (erro) { return ContentService .createTextOutput( "agroTraceCallback(" + JSON.stringify({ status: "erro", mensagem: erro.message }) + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } } // ============================================================ // BUSCAR LOTE // ============================================================ function buscarLote(qr, lote) { const planilha = SpreadsheetApp.openById(PLANILHA_ID); const aba = planilha.getSheetByName(ABA_CADASTRO); if (!aba) { throw new Error( "A aba CADASTRO_LOTES não foi encontrada." ); } const valores = aba.getDataRange().getValues(); if (valores.length < 2) { throw new Error( "Não existem lotes cadastrados." ); } const cabecalhos = valores[0]; const colunaQR = cabecalhos.findIndex(function(nome) { return String(nome) .trim() .toUpperCase() === "QR"; }); const colunaLote = cabecalhos.findIndex(function(nome) { return String(nome) .trim() .toUpperCase() === "LOTE"; }); if (colunaQR === -1) { throw new Error( "Coluna QR não encontrada." ); } if (colunaLote === -1) { throw new Error( "Coluna LOTE não encontrada." ); } qr = String(qr) .trim() .toUpperCase(); lote = String(lote) .trim() .toUpperCase(); for ( let i = 1; i < valores.length; i++ ) { const qrPlanilha = String(valores[i][colunaQR]) .trim() .toUpperCase(); const lotePlanilha = String(valores[i][colunaLote]) .trim() .toUpperCase(); const encontrouQR = qr !== "" && qrPlanilha === qr; const encontrouLote = lote !== "" && lotePlanilha === lote; if (encontrouQR || encontrouLote) { const dados = {}; for ( let j = 0; j < cabecalhos.length; j++ ) { const nomeColuna = String(cabecalhos[j]).trim(); dados[nomeColuna] = valores[i][j]; } return { status: "ok", dados: dados }; } } return { status: "nao_encontrado", mensagem: "Lote não encontrado.", lote: lote, qr: qr }; } // ============================================================ // REGISTRAR BENEFICIAMENTO // ============================================================ function registrarBeneficiamento(dados) { const planilha = SpreadsheetApp.openById(PLANILHA_ID); const aba = planilha.getSheetByName( ABA_BENEFICIAMENTOS ); if (!aba) { throw new Error( "A aba BENEFICIAMENTOS não foi encontrada." ); } const pesoAntes = Number( String(dados.pesoAntes || "") .replace(",", ".") ) || 0; const pesoDepois = Number( String(dados.pesoDepois || "") .replace(",", ".") ) || 0; const descarte = pesoAntes - pesoDepois; let rendimento = 0; if (pesoAntes > 0) { rendimento = (pesoDepois / pesoAntes) * 100; } const agora = new Date(); const id = "BEN-" + Utilities.formatDate( agora, Session.getScriptTimeZone(), "yyyyMMddHHmmss" ); const linha = [ id, dados.qr || "", dados.lote || "", dados.variedade || "", dados.peneira || "", dados.umidadeRecebimento || "", dados.pesoRecebimento || "", dados.operador || "", dados.maquina || "", dados.dataInicio || "", dados.horaInicio || "", dados.umidadeAntes || "", dados.pesoAntes || "", dados.dataFim || "", dados.horaFim || "", dados.umidadeDepois || "", dados.pesoDepois || "", descarte, rendimento, dados.tempoProcesso || "", dados.observacao || "", "FINALIZADO" ]; aba.appendRow(linha); return { status: "ok", mensagem: "Beneficiamento registrado com sucesso.", id: id }; } // ============================================================ // POST // ============================================================ function doPost(e) { try { if ( !e || !e.parameter || !e.parameter.dados ) { throw new Error( "Nenhum dado foi recebido." ); } const dados = JSON.parse( e.parameter.dados ); return respostaJSON( registrarBeneficiamento(dados) ); } catch (erro) { return respostaJSON({ status: "erro", mensagem: erro.message }); } } // ============================================================ // RESPOSTA JSON // ============================================================ function respostaJSON(objeto) { return ContentService .createTextOutput( JSON.stringify(objeto) ) .setMimeType( ContentService.MimeType.JSON ); }
